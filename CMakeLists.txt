cmake_minimum_required ( VERSION 3.10 )
project ( CrossUI C )
set ( CMAKE_C_STANDARD 11 )
set ( CMAKE_C_STANDARD_REQUIRED ON )

set ( CROSSUI_SOURCES CrossUI.c CrossUI.h CrossUI_Internal.h )

if ( WIN32 )
	list ( APPEND CROSSUI_SOURCES CrossUI_Win32.c )
	message ( STATUS "${PROJECT_NAME} is being built for Windows (Win32)" )
	add_library ( CrossUI ${CROSSUI_SOURCES} )
	target_compile_definitions ( CrossUI PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN )
	target_link_libraries ( CrossUI PUBLIC user32 gdi32 comctl32 comdlg32 shell32 ole32 )
else()
	set ( CROSSUI_LINUX_BACKEND "gtk" CACHE STRING "Linux backend: gtk or zenity" )
	set_property ( CACHE CROSSUI_LINUX_BACKEND PROPERTY STRINGS gtk zenity )
	if ( NOT CROSSUI_LINUX_BACKEND STREQUAL "zenity" )
		find_package ( PkgConfig )
		if ( PkgConfig_FOUND )
			pkg_check_modules ( GTK3 IMPORTED_TARGET gtk+-3.0 )
		endif()
	endif()
	if ( GTK3_FOUND AND NOT CROSSUI_LINUX_BACKEND STREQUAL "zenity" )
		list ( APPEND CROSSUI_SOURCES CrossUI_GTK.c )
		message ( STATUS "${PROJECT_NAME} is being built for Linux (GTK)" )
		add_library ( CrossUI ${CROSSUI_SOURCES} )
		target_link_libraries ( CrossUI PUBLIC PkgConfig::GTK3 )
	else()
		list ( APPEND CROSSUI_SOURCES CrossUI_Zenity.c )
		message ( STATUS "${PROJECT_NAME} is being built for Linux (Zenity)" )
		add_library ( CrossUI ${CROSSUI_SOURCES} )
	endif()
endif()

target_include_directories ( CrossUI PUBLIC
	${CMAKE_CURRENT_SOURCE_DIR}
	${CMAKE_CURRENT_SOURCE_DIR}/.. )

option ( CROSSUI_BUILD_DEMO "Build the CrossUI demo application" ON )
if ( CROSSUI_BUILD_DEMO )
	add_executable ( CrossUIDemo examples/demo.c )
	target_link_libraries ( CrossUIDemo PRIVATE CrossUI )
endif()
